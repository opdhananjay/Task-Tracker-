
CREATE TABLE TaskStatusHistory
(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    TaskId INT NOT NULL,

    PreviousStatus NVARCHAR(100) NULL,
    NewStatus NVARCHAR(100) NOT NULL,

    Remarks NVARCHAR(4000) NULL,

    ChangedBy INT NOT NULL,

    CreatedAtUtc DATETIME2 NOT NULL
        DEFAULT SYSUTCDATETIME()
);

(

    TestCaseTitle NVARCHAR(500) NOT NULL,
    ExpectedResult NVARCHAR(MAX) NULL,

    DeveloperActualResult NVARCHAR(MAX) NULL,
    DeveloperStatus NVARCHAR(50) NULL,

    TesterActualResult NVARCHAR(MAX) NULL,
    TesterStatus NVARCHAR(50) NULL,
    TesterRemarks NVARCHAR(4000) NULL,

    CreatedBy INT NOT NULL,
    CreatedAtUtc DATETIME2 NOT NULL
        DEFAULT SYSUTCDATETIME(),

    UpdatedBy INT NULL,
    UpdatedAtUtc DATETIME2 NULL
);
